		\section{Marker-assisted selection (trait prediction)}
		
			\textbf{Formula:}
			\begin{center}
				\boxed{	\hat{\mu} = \mu_A + \alpha_{marker} \cdot SNP}
			\end{center}
			
			\begin{itemize}
			    \item $\hat{y}$ = predicted trait value
				\item $\mu_A$ = population mean of genotype A (baseline, SNP = 0)
				\item $\alpha_{marker}$ = allele effect at the marker (B vs A)
				\item $SNP$ = genotype coding: 0 (A/A) or 1 (B)
			\end{itemize}

		\section{Breeding-Implications}
			
			\textbf{Grain production}
			\begin{itemize}
				\item TGW × RIP correlation = +0.36 (positive, moderate)
				\item So later ripening → higher TGW
				\item For the RIP markers, α is negative (SCRI\_RS\_155612 = −1.62, SCRI\_RS\_170337 = −0.59)
				\item Negative α means B allele → earlier ripening (RIP decreases from A to B)
				\item Earlier ripening → lower TGW
				\item Therefore: A is expected favor TGW (because A keeps RIP later)
			\end{itemize}
			
			\textbf{Animal Food Production}
			\begin{itemize}
				\item One might select for Marker 3 = Allele B, to get higher crops, with an acceptable effect on the other traits (from LD)
			\end{itemize}
